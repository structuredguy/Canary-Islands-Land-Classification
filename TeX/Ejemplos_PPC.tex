    \item \textbf{1. Clasificación por Áreas Temáticas y Competencias}
    \begin{itemize}
        \item \textbf{Movilidad y Accesibilidad}
        \begin{itemize}
            \item \textit{Competencia del Cabildo de Tenerife (Morado):}
            \begin{itemize}
                \item Transporte público nocturno.
                \item Transporte público más barato para reducir el tráfico.
                \item No a la Vía Exterior.
                \item Guagua circular en el Suroeste.
                \item Aceras en TF-180 Vistabella.
                \item Miradores en Anaga.
                \item Mejora de la frecuencia del transporte público.
                \item Apartaderos carreteras Anaga.
            \end{itemize}
            \item \textit{Competencia del Ayuntamiento (Azul claro):}
            \begin{itemize}
                \item Solución al estrechamiento carretera de acceso al Parque Rural por San Andrés.
                \item Conexión viaria entre barrios del Suroeste.
                \item Peatonalización de los centros urbanos.
                \item Ciudad para el peatón, no para el coche.
                \item Restringir aparcamientos en el centro SC (franja horaria).
                \item Determinar recorridos preferentes que sean accesibles a sillas de ruedas.
            \end{itemize}
        \end{itemize}

        \item \textbf{Espacio Libre Público}
        \begin{itemize}
            \item \textit{Competencia del Ayuntamiento (Azul claro):}
            \begin{itemize}
                \item Uso de barrancos como zonas verdes.
                \item No a la privatización del espacio público (terrazas).
                \item Ejecución de los Espacios Libres previstos.
                \item Rutas deportivas iluminadas por la noche.
                \item Rutas deportivas segregadas y alejadas del tráfico.
                \item Renaturalización espacios urbanos.
                \item Apertura al mar en todo el municipio.
            \end{itemize}
        \end{itemize}

        \item \textbf{Dotaciones, Equipamientos e Infraestructuras}
        \begin{itemize}
            \item \textit{Competencia del Ayuntamiento (Azul claro):}
            \begin{itemize}
                \item Pequeños intercambiadores modales en el perímetro de los barrios para dejar coche.
                \item Centros Dotacionales Integrales polivalentes.
                \item Centro de Salud en el Suroeste.
                \item Centros de participación cívica.
                \item Ejecución dotaciones previstas en Acorán. Evitar ciudad dormitorio.
                \item Mercadillo del Agricultor Suroeste.
                \item Camping entorno Las Teresitas.
            \end{itemize}
        \end{itemize}

        \item \textbf{Patrimonio Cultural}
        \begin{itemize}
            \item \textit{Competencia del Ayuntamiento (Azul claro):}
            \begin{itemize}
                \item Protección del patrimonio inmaterial.
                \item Preservación de los estanques de agua para las aves migratorias y la agricultura.
                \item Revisar el Catálogo de Protección para ajustarlo a la realidad.
                \item Protección de eras, bancales, paseros.
                \item Rehabilitación del Barranco del Muerto.
            \end{itemize}
            \item \textit{Plan Estratégico Municipal (Verde claro):}
            \begin{itemize}
                \item Fomentar comercio de proximidad.
                \item Ayudas a la Rehabilitación Edificios.
                \item Apostar por un parque de vivienda de alquiler.
                \item Divulgación del patrimonio material e inmaterial.
            \end{itemize}
        \end{itemize}

        \item \textbf{Desarrollo Sostenible}
        \begin{itemize}
            \item \textit{Competencia del Ayuntamiento (Azul claro):}
            \begin{itemize}
                \item Sistemas bio urbanos, calles orientadas a corrientes de aire para evitar aires acondicionados.
                \item Rehabilitar el paisaje de la Montaña Taco.
                \item Creación de un parque agroecológico en el Suroeste.
                \item Huertos urbanos.
                \item Separar industrias molestas del área residencial.
            \end{itemize}
            \item \textit{Plan Estratégico Municipal (Verde claro):}
            \begin{itemize}
                \item Lucha contra Cambio Climático.
                \item Fomento de las Energías Limpias.
                \item Hacer atractiva la ciudad para residir y no perder población.
            \end{itemize}
        \end{itemize}
    \end{itemize}
\end{itemize}
